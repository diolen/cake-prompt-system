	function _formatValue($type, $value, $output){
		$formattedValue = $value;
		switch($type){
			case 'percentage':
				$formattedValue = ($output === 'screen' && $value != 'N/A') ? number_format( $value, 1, ',', '').'%' : $value;
				break;
			case 'currency':
				$formattedValue = ($output === 'screen') ? number_format( $value, 0, '', '.') : $value;
				break;
			case 'date':
				$formattedValue = (new \Datetime( $value))->format('d/m/Y');
				break;
		}
		return $formattedValue;
	}

	function _formatLine($arrayLine, $output){
		$colTypes = array_values($this->_getColNames());
		$formattedLine = [];
		foreach($arrayLine as $pos => $value){
			$formattedLine[] = $this->_formatValue($colTypes[$pos], $value, $output);
		}
		return $formattedLine;
	}

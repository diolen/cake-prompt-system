                        $line = 
                        	$sic['Sic']['sic_number'] . ";" .

                        	$time->format('d/m/Y', $sic['InvoicesOc']['invoice_oc_date_expiration']) . ";" . 
                        	$sic['Center']['title'] . ";" . 
                        	$contractNumber . ";" . 
                        	$orderType . ";" . 
                        	str_replace("\r", "", $sic['Supplier']['company']) . ";" . 
                        	$item['amount'] . ";" . 
                        	$rubro . ";" . 
                        	$subrubro . ";" . 
                        	$subsubrubro . ";" . 
                        	$detail . ";" . 
                        	$item['total_price'] . ";" . 
                        	$sic['Sic']['payment_form'] . ";" . 
                        	$status . ";" . 
                        	$contractType . ";" . 
                        	str_replace("\r\n", ' ', $sic['Sic']['justification']) . ";" . 
                        	$associatedInvoice . ";" . 
                        	$nroAssociatedInvoice . ";" .

                }

        }

    }

	// chequea límite del contrato para la categoría seleccionada
	public function admin_check_limit( $contract_id, $cat_id, $amount = 0, $price = 0){

		// nueva exception para contratos que no tienen limites
		$respectLimits = true;
		$this->recursive = -1;
		$contract = $this->Sic->Contract->read( null, $contract_id);
		if( $contract){
			if( isset($contract['Contract']['no_limits']) &&  $contract['Contract']['no_limits'] == 1) $respectLimits = false;
		}
		// ----------------------------------------------------


		//$this->autoRender = false;

		// trae límites y totales del contrato relacionado
		$limits = $this->Sic->Contract->getLimits( $contract_id);
		// le paso los límites para ver como agrupa por categoría
		$totals = $this->Sic->Contract->getTotals( $contract_id, $limits);

		if( !$respectLimits){
			return true;
		}

		$this->loadModel('Category');
		$pathArray = $this->Category->getPath( $cat_id);

		// recorro el array de categorías para encontrar un límite
		$catBaseTitle = ''; // categoría tomada como referencia (última rama)
		$catBaseId = ''; // categoría tomada como referencia (última rama)
		$amountLimit = 0;
		$priceLimit = 0;
		$parentPriceLimit = 0;
		foreach( $pathArray as $cat){
			if( isset($limits[ $cat['Category']['id'] ])){
				$amountLimit = $limits[ $cat['Category']['id'] ]['amount'];
				$priceLimit = $limits[ $cat['Category']['id'] ]['price'];
				$catBaseTitle = $cat['Category']['title'];
				$catBaseId = $cat['Category']['id'];
				// parche toma el limite de la categoria padre
				if( isset($limits[ $cat['Category']['parent_id'] ]) ){
					$parentPriceLimit = $limits[ $cat['Category']['parent_id'] ]['price'];
				}
				// corta el loop a la primera 
				// break;
			}
		}


		// la categoría o su path no cuenta con límites
		if( $amountLimit == 0 || $priceLimit == 0 ){
			$this->limitError = 'El rubro seleccionado (o sus padres) no cuentan con presupuesto mensual asignado.';
			return false;
		}		

		$amountLimit = 1000000; // por cantidad no hay limite
		// para el caso que se tenga cantidad o precio

		if( $amount > 0 || $price > 0 ){

			// ahora que tengo los límites, chequeo que no se vayan a superar
			$compareAmountTotal = $amount;
			$comparePriceTotal = $price;

			// veo si tengo el total de la categoría y se lo sumo
			if( isset($totals[ $catBaseId]['amount'])){
				$compareAmountTotal += $totals[ $catBaseId ]['amount'];
			}
			if( isset($totals[ $catBaseId]['price'])){
				$comparePriceTotal += $totals[ $catBaseId ]['price'];
			}

			// saldos
			$saldoPrice = $priceLimit;
			$saldoAmount = $amountLimit;
			if( isset($totals[ $catBaseId ]['amount'])){
				$saldoAmount = $amountLimit - $totals[ $catBaseId ]['amount'];
			}
			if( isset($totals[ $catBaseId ]['price'])){
				$saldoPrice = $priceLimit - $totals[ $catBaseId ]['price'];
			}
			// fin saldos

			if( $compareAmountTotal > $amountLimit){
				$this->limitError = 'El rubro ' . $catBaseTitle . ' supera su límite de '.$amountLimit.' unidades. Saldo: '.$saldoAmount.'. Solicite un EXTRA BUDGET o RECONFIGURAR los MONTOS de cada rubro.';
				return false;			
			}
			// && $comparePriceTotal > $parentPriceLimit
			if( $comparePriceTotal > $priceLimit && $comparePriceTotal > $parentPriceLimit ){
				$this->limitError = 'El rubro ' . $catBaseTitle . ' supera el presupuesto mensual de $'.$priceLimit.'. Saldo: $'.$saldoPrice.'. Solicite un EXTRA BUDGET o RECONFIGURAR los MONTOS de cada rubro.';
				return false;			
			}		

		}

		// por las dudas no retorna
		return true;
	}

	/**
	 * Agrega un producto a una SIC
	 * Se modifico para pasar la SIC como parametro porque al parecer al usar session
	 * el mismo usuario modificaba dos SICs a la vez y se producían problemas
	 */
	public function admin_detail_add(){

		$this->autoRender = false;

		// busca la SIC en lugar de tomar la session
		$this->Sic->recursive = -1;
		$sic = $this->Sic->read(null, $this->params['data']['sic_id']);
		if(!$sic){
			$data = array( 'data' => array(), 'error' => 'Sic ID ' . $this->params['data']['sic_id'] . ' inválido' );
			die( json_encode( $data ) );	
		}

		$vehicle = '';
		$contract_id = $sic['Sic']['contract_id'];

		// verifico que los datos sean correctos y saco la cuenta del valor del item
		if( !isset($this->params['data']['cat_id']) || empty($this->params['data']['cat_id']) ){
			$data = array( 'data' => array(), 'error' => 'Seleccione un rubro' );
			die( json_encode( $data ) );			
		}
		if( !isset($this->params['data']['items_user_id']) || empty($this->params['data']['items_user_id']) ){
			$data = array( 'data' => array(), 'error' => 'Debe seleccionar un usuario del servicio/producto comprado' );
			die( json_encode( $data ) );			
		}				
		if( !isset($this->params['data']['van_id']) || empty($this->params['data']['van_id']) ){
			if(isset($this->params['data']['cat_id']) && !empty($this->params['data']['cat_id']) ) {
				$this->loadModel('Category');
				$vehiclesItems = $this->Category->isItemOfVehicle($this->params['data']['cat_id']);
				if($vehiclesItems == true) {
					$data = array( 'data' => array(), 'error' => 'Debe seleccionar un vehiculo' );
					die( json_encode( $data ) );
				}
			}
		} else {
	        $this->loadModel('Van');
	        $vehicles = $this->Van->getList();
	        if(!empty($this->params['data']['van_id'])) {
	        	$vehicle = ' ' . $vehicles[$this->params['data']['van_id']];
	        }			
		}
		if( !isset($this->params['data']['item_amount']) || !is_numeric($this->params['data']['item_amount']) ){
			$data = array( 'data' => array(), 'error' => 'Ingrese una cantidad válida' );
			die( json_encode( $data ) );			
		}
		if( !isset($this->params['data']['item_id']) || ! $item = $this->Sic->SicsItem->Item->read(null, $this->params['data']['item_id']) ){
			$data = array( 'data' => array(), 'error' => 'Debe seleccionar un Item' );
			die( json_encode( $data ) );
		}else{
			// precio
			$priceEstimated = (isset($this->params['data']['item_price']) && is_numeric($this->params['data']['item_price']) && $this->params['data']['item_price']*1 > 0)?$this->params['data']['item_price']:$item['Item']['price'];
			// saco el precio total que sería 
			$totalPrice = $priceEstimated * $this->params['data']['item_amount'];
		}

		if( isset($this->params['data']['item_detail']) && !empty($this->params['data']['item_detail'])){
			$detail = $this->params['data']['item_detail'];
		}else{
			$detail = $item['Item']['title'];
		}

		// veo límites del arbol
		if( $this->admin_check_limit( $contract_id, $this->params['data']['cat_id'], $this->params['data']['item_amount'], $totalPrice) || $contract_id==0  ){		

			// agrego el item
			$dataToSave = array(
				'sic_id' => $sic['Sic']['id'],
				'item_id' => $this->params['data']['item_id'],
				'van_id' => isset($this->params['data']['van_id']) ? $this->params['data']['van_id'] : '',
				'amount' => $this->params['data']['item_amount'],
				'unit_price' => $priceEstimated,
				'item_iva' => $this->params['data']['item_iva'],
				'total_price' => $totalPrice,
				'detail' => $detail . $vehicle,
				'robinson' => 0,
				// auditoría
				'log_user_id' => $this->Session->read('Auth.User.id')
				);

			$this->Sic->SicsItem->create($dataToSave);
			if( $this->Sic->SicsItem->save()){
				//Si es un "contrato", es decir, no es "contrato eventual"
				if($contract_id !=0){
					// trae límites y totales del contrato relacionado
					$limits = $this->Sic->Contract->getLimits( $contract_id, '', true);
					// le paso los límites para ver como agrupa por categoría
					$totals = $this->Sic->Contract->getTotals( $contract_id, $limits, '', true);
					// --------------
					$data = array( 'data' => array(), 'error' => '' );
				}else{
					//Si es un "contrato eventual", resultado OK. con esto agrega los items
					$data = array( 'data' => array(), 'error' => '' );
				}
				$this->Sic->saveField( 'items_user_id', $this->params['data']['items_user_id']);
			}else{
				$data = array( 'data' => array(), 'error' => 'Error al guardar' );
			}
			die( json_encode( $data ) );

		}else{
			$data = array( 'data' => array(), 'error' => $this->limitError );
			die( json_encode( $data ) );
		}

	}

	public function updateTotalCost($id = false)
	{
		$evt = $this->EventualContract->read(null, $id);
		
		$options = array(
			'fields' => array('sum(Sic.total) as sictotal'),
			'conditions' => array('Sic.closed' => 1,'Sic.robinson'=>0,'Sic.contract_type_id'=>1,'Sic.eventual_contract_id'=>$id)
		);
		
		$sictotal = $this->EventualContract->Sic->find('first', $options);
		$monto_sic = $sictotal[0]['sictotal'];
		if(!isset($monto_sic)){$monto_sic=0;}
		
		$total_cost = $monto_sic+
					$evt['EventualContract']['labor']+
					$evt['EventualContract']['fixedcost']+
					$evt['EventualContract']['othercost'];
					
		$dataToUpdate = array('id' => $id,'cost' => $monto_sic, 'total_cost' => $total_cost);
		$this->EventualContract->save( $dataToUpdate, false);
		
		return;
	}

		$jsId = $this->Session->read('App.Sic.user_id');
	
		// Info completa de UO
		$this->loadModel('UsersCenter');
		$this->loadModel('OperationUnit');
		$this->loadModel('UsersOperationUnit');
		$operarionUnitsInfo = $this->UsersOperationUnit->getOperationUnitsInfo(
			$this->UsersCenter,
			$this->OperationUnit,
			$this->Sic->Center,
			$authUserId
		);
	
		$userCenters = $operarionUnitsInfo['userCenters'];
	
		$this->set('userCenters', $userCenters);
		$this->set('centersGT', $operarionUnitsInfo['centersGT']);
		$this->set('centersTO', $operarionUnitsInfo['centersTO']);
		$this->set('usersOperationUnit', $operarionUnitsInfo['usersOperationUnit']);
		$this->set('operationCentersUnits', $operarionUnitsInfo['operationCentersUnits']);
		$this->set('operationUnits', $operarionUnitsInfo['operationUnits']);
		// Fin de Info de UO
	
		if ($role == 10 && (empty($joId) || empty($jsId))) {
			if (empty($this->Session->read('App.Sic.center_id'))) {
				if (count($userCenters) > 0) {
					$this->paginate['conditions']['Sic.center_id'] = $userCenters;
				} else {
					$this->paginate['conditions']['Sic.center_id'] = $this->Auth->user('center_id');
				}
			}
		}
	
		$this->applyFilters();
		$centerId = $this->Session->read('App.Sic.center_id');
		$locationTitle = $this->Session->read('App.Sic.location');
		$eventualContractId = $this->Session->read('App.Sic.eventual_contract_id');
	
		if ($role == 60 || !empty($eventualContractId) || !empty($joId)) {
			$this->paginate['joins'] = array(
				array(
					'table' => 'eventual_contract_budgets',
					'alias' => 'EventualContractBudget',
					'type' => 'INNER',
					'conditions' => array(array('Sic.eventual_contract_id = EventualContractBudget.id'))
				)
			);
			$this->paginate['conditions']['Sic.contract_type_id'] = 1;
	
			if ($role == 60) {
				$this->paginate['conditions']['EventualContractBudget.jefeobra_id'] = $authUserId;
				if (count($userCenters) > 0) {
					$this->paginate['conditions']['EventualContractBudget.center_id'] = $userCenters;
				}
			}
			if (!empty($joId)) {
				$this->paginate['conditions']['EventualContractBudget.jefeobra_id'] = $joId;
			}
			if (!empty($eventualContractId)) {
				$this->paginate['conditions']['EventualContractBudget.id'] = $eventualContractId;
			}
		} elseif ($role == 20) {
			if (empty($centerId)) {
				if (count($userCenters) > 0) {
					$this->paginate['conditions']['Sic.center_id'] = $userCenters;
				} else {
					$this->paginate['conditions']['Sic.center_id'] = $this->Auth->user('center_id');
				}
			}
			$this->paginate['conditions']['Sic.contract_type_id'] = 0;
		}
	
		if (!empty($locationTitle)) {
			$this->paginate['joins'] = array(
				array(
					'table' => 'eventual_contract_budgets',
					'alias' => 'EventualContractBudget',
					'type' => 'left',
					'conditions' => array('Sic.eventual_contract_id = EventualContractBudget.id')
				)
			);
			$this->paginate['conditions']['EventualContractBudget.location'] = $locationTitle;
			$this->request->data['Search']['location'] = $locationTitle;
		}
	
		if (!empty($jsId)) {
			$this->paginate['conditions']['Contract.user_id'] = $jsId;
			$this->paginate['conditions']['Sic.contract_type_id'] = 0;
		}
	
		$this->paginate['conditions']['Sic.robinson <>'] = 1;
		$this->paginate['order'] = 'Sic.date_of DESC';
		$this->paginate['limit'] = 500;
		$this->paginate['fields'] = array(
			'Sic.id',
			'Sic.sic_number',
			'Sic.date_of',
			'Sic.type_id',
			'Sic.closed',
			'Sic.payment_form',
			'Sic.justification',
			'Center.id',
			'Center.title',
			'LogUser.id',
			'LogUser.first_name',
			'Supplier.id',
			'Supplier.company',
			'Contract.id',
			'Contract.type_id',
			'Contract.contract_number',
			'InvoicesOc.id',
			'InvoicesOc.finvoice_oc_file_name',
			'InvoicesOc.invoice_oc_branch',
			'InvoicesOc.invoice_oc_number',
			'InvoicesOc.invoice_oc_date_expiration'
		);
		$this->paginate['contain'] = array(
			'Center' => array('fields' => array('Center.id', 'Center.title')),
			'LogUser' => array('fields' => array('LogUser.id', 'LogUser.first_name')),
			'Supplier' => array('fields' => array('Supplier.id', 'Supplier.company')),
			'Contract' => array('fields' => array('Contract.id', 'Contract.type_id', 'Contract.contract_number')),
			'InvoicesOc' => array('fields' => array(
				'InvoicesOc.id',
				'InvoicesOc.finvoice_oc_file_name',
				'InvoicesOc.invoice_oc_branch',
				'InvoicesOc.invoice_oc_number',
				'InvoicesOc.invoice_oc_date_expiration'
			))
		);
		$this->Sic->Behaviors->attach('Containable');
		$this->Sic->SicsItem->Behaviors->attach('Containable');
	
		// Agregar el filtro de fecha para los sic (se conserva comportamiento original)
		if (isset($this->request->data['Search']['sic_date_from']) && isset($this->request->data['Search']['sic_date_to'])) {
			$this->Session->write('App.Sic.sic_date_from', $this->request->data['Search']['sic_date_from']);
			$this->Session->write('App.Sic.sic_date_to', $this->request->data['Search']['sic_date_to']);
		}
	
		echo utf8_decode("SIC #;Fecha;Fecha De Vencimiento;Centro;Contrato;Tipo;Proveedor;Cantidad;Rubro;Subrubro;Item;Detalle;Importe;Condicion De Pago;Estado;Tipo De Contrato;Justificación;Factura Asociada;Nro.Factura;Generado Por;") . "\n";
	
		$baseQuery = $this->paginate;
		$page = 1;
	
		while (true) {
			$query = $baseQuery;
			$query['page'] = $page;
	
			$sics = $this->Sic->find('all', $query);
			if (empty($sics)) {
				break;
			}
	
			$sicIds = array();
			foreach ($sics as $sic) {
				$sicIds[] = $sic['Sic']['id'];
			}
	
			$sicsItemsRows = array();
			if (!empty($sicIds)) {
				$sicsItemsRows = $this->Sic->SicsItem->find('all', array(
					'fields' => array(
						'SicsItem.id',
						'SicsItem.sic_id',
						'SicsItem.amount',
						'SicsItem.total_price',
						'SicsItem.detail',
						'Item.id',
						'Item.path',
						'Item.title'
					),
					'conditions' => array(
						'SicsItem.sic_id' => $sicIds,
						'SicsItem.robinson <>' => 1
					),
					'contain' => array(
						'Item' => array('fields' => array('Item.id', 'Item.path', 'Item.title'))
					),
					'order' => array('SicsItem.id' => 'ASC')
				));
			}
	
			$itemsBySic = array();
			foreach ($sicsItemsRows as $row) {
				$sid = $row['SicsItem']['sic_id'];
				if (!isset($itemsBySic[$sid])) {
					$itemsBySic[$sid] = array();
				}
				$itemsBySic[$sid][] = $row;
			}
	
			foreach ($sics as $sic) {
				$sid = $sic['Sic']['id'];
				if (empty($itemsBySic[$sid])) {
					continue;
				}
	
				$status = ($sic['Sic']['closed'] == 1) ? 'Aprobada' : 'En proceso';
				$logUser = isset($sic['LogUser']['first_name']) ? $sic['LogUser']['first_name'] : '-';
				$contractType = (isset($this->_contract_types[$sic['Contract']['type_id']])) ? $this->_contract_types[$sic['Contract']['type_id']] : '-';
				$contractNumber = (!empty($sic['Contract']['contract_number'])) ? $sic['Contract']['contract_number'] : '-';
				$associatedInvoice = isset($sic['InvoicesOc']['finvoice_oc_file_name']) ? $sic['InvoicesOc']['finvoice_oc_file_name'] : '-';
				$nroAssociatedInvoice = (isset($sic['InvoicesOc']['invoice_oc_branch']) && isset($sic['InvoicesOc']['invoice_oc_number']))
					? $sic['InvoicesOc']['invoice_oc_branch'] . '-' . $sic['InvoicesOc']['invoice_oc_number']
					: '-';
				$orderType = (isset($this->_order_types[$sic['Sic']['type_id']])) ? $this->_order_types[$sic['Sic']['type_id']] : '-';
	
				$dateOf = (!empty($sic['Sic']['date_of'])) ? $time->format('d/m/Y', $sic['Sic']['date_of']) : '-';
				$expirationDate = (!empty($sic['InvoicesOc']['invoice_oc_date_expiration']))
					? $time->format('d/m/Y', $sic['InvoicesOc']['invoice_oc_date_expiration'])
					: '-';
	
				$centerTitle = isset($sic['Center']['title']) ? $sic['Center']['title'] : '-';
				$supplier = isset($sic['Supplier']['company']) ? str_replace("\r", "", $sic['Supplier']['company']) : '-';
				$paymentForm = isset($sic['Sic']['payment_form']) ? $sic['Sic']['payment_form'] : '-';
				$justification = isset($sic['Sic']['justification']) ? str_replace("\r\n", ' ', $sic['Sic']['justification']) : '-';
	
				foreach ($itemsBySic[$sid] as $itemRow) {
					$item = $itemRow['SicsItem'];
					$itemInfo = isset($itemRow['Item']) ? $itemRow['Item'] : array();
	
					$path = '->->->';
					if (isset($itemInfo['path']) && !empty($itemInfo['path'])) {
						$path = $itemInfo['path'];
					}
	
					$splitPath = explode('>', $path);
					$rubro = isset($splitPath[0]) ? $splitPath[0] : '-';
					$subrubro = isset($splitPath[1]) ? $splitPath[1] : '-';
					$subsubrubro = isset($itemInfo['title']) ? $itemInfo['title'] : '-';
					$detail = isset($item['detail']) ? str_replace(';', ' ', $item['detail']) : '-';
	
					$line =
						$sic['Sic']['sic_number'] . ";" .
						$dateOf . ";" .
						$expirationDate . ";" .
						$centerTitle . ";" .
						$contractNumber . ";" .
						$orderType . ";" .
						$supplier . ";" .
						$item['amount'] . ";" .
						$rubro . ";" .
						$subrubro . ";" .
						$subsubrubro . ";" .
						$detail . ";" .
						$item['total_price'] . ";" .
						$paymentForm . ";" .
						$status . ";" .
						$contractType . ";" .
						$justification . ";" .
						$associatedInvoice . ";" .
						$nroAssociatedInvoice . ";" .
						$logUser . ";";
	
					echo utf8_decode($line), "\n";
				}
			}
	
			@ob_flush();
			flush();
	
			unset($sics, $sicIds, $sicsItemsRows, $itemsBySic);
			$page++;
		}
	
		die();
	}

	/**
	 * Método para procesar y filtrar los Sics que necesitan calificación (sin rating previo)
	 */
	private function processSicsForRating() {
		// Definir la fecha de inicio para el rating (1 año atrás)
		$ratingSuppliersStartDate = (date('Y') - 1) . "-" . date('m-d');

		// Cargar el modelo RatingSupplier
		$this->loadModel('RatingSupplier');

		// Obtener los sic_ids que ya tienen rating (como array plano para lookup O(1))
		$optionsRatingSuppliers = array(
			'fields' => array('RatingSupplier.sic_id'),
			'conditions' => array(
				'RatingSupplier.created >' => $ratingSuppliersStartDate
			),
			'group' => array('RatingSupplier.sic_id'),
			'recursive' => -1
		);

		$ratingsResult = $this->RatingSupplier->find('all', $optionsRatingSuppliers);
		$ratingsWithSicIds = array();
		foreach ($ratingsResult as $rating) {
			$ratingsWithSicIds[$rating['RatingSupplier']['sic_id']] = true;
		}

		// Opciones para obtener los Sics que deben procesarse, excluyendo los que ya tienen rating
		$optionsSicsToProcess = $this->paginate;
		unset($optionsSicsToProcess['contain']);
		unset($optionsSicsToProcess['limit']);
		unset($optionsSicsToProcess['page']);
		$optionsSicsToProcess['fields'] = array('Sic.id', 'Sic.supplier_id');
		$optionsSicsToProcess['conditions']['Sic.date_of >='] = $ratingSuppliersStartDate;
		
		// Excluir Sics que ya tienen rating usando NOT IN
		if (!empty($ratingsWithSicIds)) {
			$optionsSicsToProcess['conditions']['NOT'] = array('Sic.id' => array_keys($ratingsWithSicIds));
		}
		
		$optionsSicsToProcess['recursive'] = -1;

		// Obtener los Sics filtrados directamente en la consulta
		$sicsToProcess = $this->Sic->find('all', $optionsSicsToProcess);

		// Construir el array resultado
		$filteredSicsToProcess = array();
		foreach ($sicsToProcess as $sic) {
			$filteredSicsToProcess[$sic['Sic']['id']] = $sic['Sic']['supplier_id'];
		}
		return $filteredSicsToProcess;
	}

    public function maxDaysLimit($check, $limit) {

        if(!empty($check['date_of'])) {
            $now = time();
            $your_date = strtotime($check['date_of']);
            $datediff = $now - $your_date;
            $days = round($datediff / (60 * 60 * 24));

            if($days > $limit) {
                return false;
            }
        } else {
            return false;
        }
        
        return true;
    }

    public function checkSupplierData($check) {
        $supplier = $this->query("SELECT * FROM suppliers WHERE id = " .$check['supplier_id']);

        if(count($supplier) > 0) {
            if($supplier[0]['suppliers']['iibb_category_id'] == '' || $supplier[0]['suppliers']['iibb_category_id'] == 0 || $supplier[0]['suppliers']['iva_id'] == '' || $supplier[0]['suppliers']['iva_id'] == 0) {
                return false;
            } elseif($supplier[0]['suppliers']['bank_name'] == '' || $supplier[0]['suppliers']['bank_cbu'] == '' || $supplier[0]['suppliers']['bank_account'] == '' || $supplier[0]['suppliers']['bank_accout_name'] == '' || $supplier[0]['suppliers']['contact_email'] == '') {
                return false;
            }
        }
        return true;
    }

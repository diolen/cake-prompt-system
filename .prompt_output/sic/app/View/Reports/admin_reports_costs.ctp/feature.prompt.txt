$(function() {

        $.ajax({

            dataType: 'json',
            url: '<?php echo Router::url(array('controller' => 'Reports', 'action' => 'getCenters')); ?>',
            data: {'operationUnitId': operationUnitId}

            if(data) {
                $('#divSearchCenterId').html('');

                    output += '<option value="'+k+'">'+v+'</optiion>';
                });
                output += '</select>';
                $('#divSearchCenterId').html(output);
                $('#SearchCenterId').val($.parseJSON(centerId));
                $('.chosen-select').chosen();
            }
        });        
    }    

    // Load centers if operation units are pre-selected
    if(operationUnitId.length > 0) {
        centerId = '<?php echo !empty($filterCenterId) ? json_encode($filterCenterId) : 0; ?>';
        getCenters(operationUnitId, centerId);        
    }

    // Reload centers when operation unit selection changes
    $('#SearchOperationUnitId').on('change click', function() {
        operationUnitId = $(this).val();
        getCenters(operationUnitId, centerId);
    });
});
